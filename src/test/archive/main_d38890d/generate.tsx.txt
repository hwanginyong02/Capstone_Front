/** main 의 useIssuance.applyIssuance 와 같은 몸(그 파일 안 비공개 함수라 옮겨 적음). 발급 뒤 Report·인쇄 화면이 그리던 데이터. */
function applyIssuance(report: any, r: any) {
  return {
    ...report,
    meta: { ...report.meta, reportId: r.reportNo, issuedAt: formatKstDateTime(r.issuedAt) },
    performer: {
      orgName: r.organization.orgName || report.performer.orgName,
      evaluator: r.organization.evaluator ?? report.performer.evaluator,
      contact: r.organization.contact ?? report.performer.contact,
    },
    signature: {
      issuer: r.issuer,
      signedAt: formatKstDate(r.issuedAt),
      history: r.history.map((h: any) => ({ version: h.version, issuedAt: formatKstDate(h.issuedAt), note: h.note ?? "" })),
    },
  };
}

const BLOCK = new Set(["H1", "H2", "H3", "H4", "H5", "P", "LI", "TR", "BLOCKQUOTE", "DT", "DD", "SECTION", "DIV", "TH", "TD"]);
function blockText(nodes: Node[], options: { dropPrintHidden?: boolean } = {}): string {
  const lines: string[] = [];
  let current = "";
  const flush = () => { const l = current.replace(/\s+/g, " ").trim(); if (l) lines.push(l); current = ""; };
  const walk = (node: Node) => {
    if (node.nodeType === Node.TEXT_NODE) { current += node.textContent ?? ""; return; }
    if (!(node instanceof Element)) return;
    if (node.tagName === "STYLE" || node.tagName === "SCRIPT") return;
    if (options.dropPrintHidden && node.classList.contains("print:hidden")) return;
    const block = BLOCK.has(node.tagName);
    if (block) flush();
    if (node.tagName === "TD" || node.tagName === "TH") { current += " | "; node.childNodes.forEach(walk); return; }
    node.childNodes.forEach(walk);
    if (block) flush();
  };
  nodes.forEach(walk);
  flush();
  return `${lines.join("\n")}\n`;
}
