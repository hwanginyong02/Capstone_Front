const BLOCK = new Set(["H1", "H2", "H3", "H4", "H5", "P", "LI", "TR", "BLOCKQUOTE", "DT", "DD", "SECTION", "DIV", "TH", "TD"]);
function blockText(nodes: Node[], options: { dropPrintHidden?: boolean } = {}): string {
  const lines: string[] = [];
  let current = "";
  const flush = () => { const l = current.replace(/\s+/g, " ").trim(); if (l) lines.push(l); current = ""; };
  const walk = (node: Node) => {
    if (node.nodeType === Node.TEXT_NODE) { current += node.textContent ?? ""; return; }
    if (!(node instanceof Element)) return;
    if (node.tagName === "STYLE" || node.tagName === "SCRIPT") return;
    if (options.dropPrintHidden && node.classList.contains("print:hidden")) return;
    const block = BLOCK.has(node.tagName);
    if (block) flush();
    if (node.tagName === "TD" || node.tagName === "TH") { current += " | "; node.childNodes.forEach(walk); return; }
    node.childNodes.forEach(walk);
    if (block) flush();
  };
  nodes.forEach(walk);
  flush();
  return `${lines.join("\n")}\n`;
}

      seen.request = JSON.parse(String(init!.body));
      if (mode === "requests") return json({ detail: "멈춤" }, 503);
      return json(readJson(`${WORK}/${spec.name}.job_created.json`));
    }
    if (method === "GET" && /\/api\/narrative\/jobs\/[^/]+$/.test(url)) {
      const created = readJson(`${WORK}/${spec.name}.job_created.json`);
      if (spec.end === "done") return json(readJson(`${WORK}/${spec.name}.job_view.json`));
      if (spec.end === "pending") {
        return json({ job_id: created.job_id, status: "running", stage: "generating", failure: null, benchmark_refs: created.benchmark_refs, result: null });
      }
      return json({ job_id: created.job_id, status: "failed", stage: "failed", failure: "error", benchmark_refs: created.benchmark_refs, result: null });
    }
    throw new Error(`생성기: 모르는 요청 ${method} ${url}`);
  }));
}

 * 정답 글
 * ------------------------------------------------------------------ */
interface Target {
  name: string;
  content: any;
  issuance: any | null;
  /** 번호로 연 화면의 서버 응답(내용 제외). */
  stored: { reportNo: string; version: string; issuedAt: string; contentHash: string };
  /** /report/:id 페이지 상태. 없으면 그리지 않는다. */
  page: null | { reportData: any; narrative: any };
}

function targets(): Target[] {
  const list: Target[] = [];
  for (let n = 1; n <= 6; n += 1) {
    const no = `RPT-2026-000${n}`;
    const raw = readFileSync(`${ISSUED}/${no}.content.json`, "utf8");
    const issuance = readJson(`${ISSUED}/${no}.issuance.json`);
    const content = JSON.parse(raw);
    list.push({
      name: no, content, issuance,
      stored: { reportNo: no, version: issuance.version, issuedAt: issuance.issuedAt, contentHash: createHash("sha256").update(raw, "utf8").digest("hex") },
      page: { reportData: applyIssuance(content, issuance), narrative: { status: "done", jobId: `job-${no}` } },
    });
  }
  const fake = { reportNo: "RPT-2026-0001", version: "v1.0", issuedAt: "2026-10-01T03:00:00+00:00", contentHash: "a".repeat(64) };
  for (const name of ["schema3_binary", "schema3_binary_pass", "schema2_binary", "schema2_multiclass_noproblems"]) {
    list.push({ name, content: readJson(`${ARCHIVE}/${name}.json`), issuance: null, stored: fake, page: null });
  }
  for (const spec of SYN) {
    const content = readJson(`${ISSUED}/${spec.name}.json`);
    const narrative = readJson(`${ISSUED}/${spec.name}.narrative.json`);
    list.push({
      name: spec.name, content, issuance: null, stored: fake,
      page: spec.end === "done" ? null : { reportData: content, narrative },
    });
  }
  return list;
}
